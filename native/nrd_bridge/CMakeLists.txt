cmake_minimum_required(VERSION 3.25)
project(bedray_nrd_bridge LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The NRD bridge currently uses the Windows DLL loader")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BEDRAY_VULKAN_HEADERS "" CACHE PATH "Vulkan-Headers include directory")
set(BEDRAY_NRD_HEADERS "" CACHE PATH "NRD Include directory matching the runtime SDK")
if(NOT EXISTS "${BEDRAY_VULKAN_HEADERS}/vulkan/vulkan.h" OR NOT EXISTS "${BEDRAY_NRD_HEADERS}/NRD.h")
    message(FATAL_ERROR "Set BEDRAY_VULKAN_HEADERS and BEDRAY_NRD_HEADERS")
endif()
enable_testing()
add_library(vkb_nrd SHARED src/vkb_nrd.cpp)
target_include_directories(vkb_nrd PRIVATE include "${BEDRAY_VULKAN_HEADERS}" "${BEDRAY_NRD_HEADERS}")
target_link_libraries(vkb_nrd PRIVATE user32)
add_executable(nrd_probe probe/nrd_probe.cpp)
target_include_directories(nrd_probe PRIVATE include "${BEDRAY_VULKAN_HEADERS}")
target_link_libraries(nrd_probe PRIVATE vkb_nrd)
add_executable(nrd_primary_lobe_settings_test tests/primary_lobe_settings_test.cpp)
target_include_directories(nrd_primary_lobe_settings_test PRIVATE src "${BEDRAY_NRD_HEADERS}")
add_test(NAME nrd_primary_lobe_settings COMMAND nrd_primary_lobe_settings_test)
if(MSVC)
    foreach(nrd_target vkb_nrd nrd_probe nrd_primary_lobe_settings_test)
        target_compile_options(${nrd_target} PRIVATE /W3 /EHsc /utf-8)
        set_property(TARGET ${nrd_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
endif()
