cmake_minimum_required(VERSION 3.25)
project(bedray_gpu_resources LANGUAGES C CXX)

set(BEDRAY_VULKAN_HEADERS "" CACHE PATH "Vulkan-Headers include directory")
if(NOT EXISTS "${BEDRAY_VULKAN_HEADERS}/vulkan/vulkan.h")
    message(FATAL_ERROR "Set BEDRAY_VULKAN_HEADERS to the Vulkan-Headers include directory")
endif()
add_library(bedray_gpu_resources STATIC src/gpu_buffers.cpp src/gpu_images.cpp)
target_include_directories(bedray_gpu_resources PUBLIC include "${BEDRAY_VULKAN_HEADERS}")
target_compile_features(bedray_gpu_resources PUBLIC cxx_std_20)
set_property(TARGET bedray_gpu_resources PROPERTY POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set_property(TARGET bedray_gpu_resources PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(bedray_gpu_resources PRIVATE /W4 /permissive- /EHsc)
endif()

add_library(bedray_gpu_resources_c STATIC src/gpu_buffers_c.cpp src/gpu_images_c.cpp)
target_link_libraries(bedray_gpu_resources_c PUBLIC bedray_gpu_resources)
set_property(TARGET bedray_gpu_resources_c PROPERTY POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set_property(TARGET bedray_gpu_resources_c PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(bedray_gpu_resources_c PRIVATE /W4 /permissive- /EHsc)
endif()

# Standalone ABI library for export/layout verification. This is not the
# bedray_core DLL; the core integration must explicitly export the wrapper.
option(BEDRAY_GPU_RESOURCES_SHARED_ABI "Build a standalone C ABI DLL" OFF)
if(BEDRAY_GPU_RESOURCES_SHARED_ABI)
    add_library(bedray_gpu_resources_abi SHARED src/gpu_buffers_c.cpp src/gpu_images_c.cpp)
    target_compile_definitions(bedray_gpu_resources_abi PRIVATE BEDRAY_GPU_RESOURCES_EXPORTS)
    target_link_libraries(bedray_gpu_resources_abi PRIVATE bedray_gpu_resources)
    if(MSVC)
        set_property(TARGET bedray_gpu_resources_abi PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()

option(BEDRAY_GPU_RESOURCES_TESTS "Build isolated mock-driver tests" OFF)
if(BEDRAY_GPU_RESOURCES_TESTS)
    enable_testing()
    add_executable(bedray_gpu_images_test tests/gpu_images_test.cpp)
    target_link_libraries(bedray_gpu_images_test PRIVATE bedray_gpu_resources_c)
    add_test(NAME bedray_gpu_images COMMAND bedray_gpu_images_test)
    add_executable(bedray_gpu_buffers_test tests/gpu_buffers_test.cpp)
    target_link_libraries(bedray_gpu_buffers_test PRIVATE bedray_gpu_resources_c)
    add_test(NAME bedray_gpu_buffers COMMAND bedray_gpu_buffers_test)
    add_executable(bedray_gpu_c_header_test tests/gpu_c_header_test.c)
    target_include_directories(bedray_gpu_c_header_test PRIVATE include)
    add_test(NAME bedray_gpu_c_header COMMAND bedray_gpu_c_header_test)
    if(BEDRAY_GPU_RESOURCES_SHARED_ABI)
        add_executable(bedray_gpu_dynamic_c_test tests/gpu_dynamic_c_test.c)
        target_include_directories(bedray_gpu_dynamic_c_test PRIVATE include)
        target_link_libraries(bedray_gpu_dynamic_c_test PRIVATE bedray_gpu_resources_abi)
    endif()
    if(MSVC)
        set_property(TARGET bedray_gpu_buffers_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        set_property(TARGET bedray_gpu_images_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
