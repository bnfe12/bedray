cmake_minimum_required(VERSION 3.24)
project(bedray_vendor_resources LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Vendor wrapper verification requires Windows x64 MSVC")
endif()
set(VKB_NGX_SDK "" CACHE PATH "Pinned NVIDIA DLSS SDK root")
set(VKB_NGX_NR_HEADERS "" CACHE PATH "Optional NVIDIA RTX Remix DLSS-NR overlay include directory")
set(VKB_SL_SDK "" CACHE PATH "Pinned Streamline SDK root")
set(BEDRAY_VULKAN_HEADERS "" CACHE PATH "Vulkan-Headers include directory")
foreach(required "${VKB_NGX_SDK}/include/nvsdk_ngx_helpers_dlssd_vk.h" "${VKB_NGX_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib" "${VKB_SL_SDK}/include/sl.h" "${BEDRAY_VULKAN_HEADERS}/vulkan/vulkan.h")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing SDK input: ${required}")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded$<$<CONFIG:Debug>:Debug>)
enable_testing()
set(WRAPPERS "${CMAKE_CURRENT_LIST_DIR}/..")
function(wrapper_settings target)
  target_include_directories(${target} PRIVATE "${WRAPPERS}/dlss_bridge/include" "${VKB_NGX_SDK}/include" "${VKB_SL_SDK}/include" "${BEDRAY_VULKAN_HEADERS}" "${CMAKE_CURRENT_LIST_DIR}")
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
endfunction()
set(NGX_SOURCES "${WRAPPERS}/dlss_bridge/src/vkb_dlss.cpp" "${WRAPPERS}/dlss_bridge/src/vkb_dlssnr.cpp")
set(NGX_LIBRARY "${VKB_NGX_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
# These are actual release wrapper builds, linked against the genuine SDK static library.
add_library(vendor_ngx_production SHARED ${NGX_SOURCES})
wrapper_settings(vendor_ngx_production)
target_link_libraries(vendor_ngx_production PRIVATE "${NGX_LIBRARY}" user32 advapi32)
set_target_properties(vendor_ngx_production PROPERTIES OUTPUT_NAME vkb_dlss RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/production/ngx" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/production/ngx")
if(VKB_NGX_NR_HEADERS)
  foreach(header nvsdk_ngx_defs_dlssnr.h nvsdk_ngx_helpers_dlssnr_vk.h)
    if(NOT EXISTS "${VKB_NGX_NR_HEADERS}/${header}")
      message(FATAL_ERROR "Missing Neural Rendering SDK header: ${header}")
    endif()
  endforeach()
  set(merged "${CMAKE_CURRENT_BINARY_DIR}/ngx-nr-include")
  file(COPY "${VKB_NGX_SDK}/include/" DESTINATION "${merged}")
  file(COPY "${VKB_NGX_NR_HEADERS}/" DESTINATION "${merged}")
  add_library(vendor_ngx_nr SHARED ${NGX_SOURCES})
  wrapper_settings(vendor_ngx_nr)
  target_include_directories(vendor_ngx_nr BEFORE PRIVATE "${merged}")
  target_compile_definitions(vendor_ngx_nr PRIVATE VKB_DLSS_ENABLE_NR=1)
  target_link_libraries(vendor_ngx_nr PRIVATE "${NGX_LIBRARY}" user32 advapi32)
  set_target_properties(vendor_ngx_nr PROPERTIES OUTPUT_NAME vkb_dlss_nr_contract RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/partner")
endif()
add_library(vendor_sl_production SHARED "${WRAPPERS}/sl_bridge/src/vkb_sl.cpp" "${WRAPPERS}/sl_bridge/src/vkb_sl_dlss.cpp")
wrapper_settings(vendor_sl_production)
target_link_libraries(vendor_sl_production PRIVATE wintrust crypt32 user32)
set_target_properties(vendor_sl_production PROPERTIES OUTPUT_NAME vkb_sl RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/production/sl" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/production/sl")
# Same production entrypoints and real SDK headers/helpers; only the SDK boundary is substituted.
foreach(provider ngx sl)
  if(provider STREQUAL ngx)
    set(source ${NGX_SOURCES})
  else()
    set(source "${WRAPPERS}/sl_bridge/src/vkb_sl_dlss.cpp")
  endif()
  add_library(vendor_${provider}_fault SHARED ${source} "${provider}_fake.cpp")
  wrapper_settings(vendor_${provider}_fault)
  target_compile_definitions(vendor_${provider}_fault PRIVATE VENDOR_TEST_BACKEND_BUILD)
  add_executable(vendor_${provider}_contract entrypoints_test.cpp)
  wrapper_settings(vendor_${provider}_contract)
  target_compile_definitions(vendor_${provider}_contract PRIVATE VENDOR_TEST_${provider})
  target_link_libraries(vendor_${provider}_contract PRIVATE vendor_${provider}_fault)
  add_test(NAME vendor_${provider}_entrypoints COMMAND vendor_${provider}_contract)
endforeach()
