cmake_minimum_required(VERSION 3.25)
project(bedray_render_graph LANGUAGES CXX)

add_library(bedray_render_graph STATIC src/render_graph.cpp src/semantic_bindings.cpp)
set_property(TARGET bedray_render_graph PROPERTY POSITION_INDEPENDENT_CODE ON)
target_include_directories(bedray_render_graph PUBLIC include)
target_compile_features(bedray_render_graph PUBLIC cxx_std_20)
if(MSVC)
    set_property(TARGET bedray_render_graph PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# The native core defines the history target first. A standalone graph build
# can supply that target separately when the submission bridge is needed.
if(TARGET bedray_history)
    add_library(bedray_graph_history STATIC src/history_submission.cpp)
    set_property(TARGET bedray_graph_history PROPERTY POSITION_INDEPENDENT_CODE ON)
    target_include_directories(bedray_graph_history PUBLIC include)
    target_compile_features(bedray_graph_history PUBLIC cxx_std_20)
    target_link_libraries(bedray_graph_history PUBLIC bedray_render_graph bedray_history)
    if(MSVC)
        set_property(TARGET bedray_graph_history PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    option(BEDRAY_GRAPH_BUILD_TESTS "Build focused graph/history tests" OFF)
    if(BEDRAY_GRAPH_BUILD_TESTS)
        add_executable(bedray_history_submission_test tests/history_submission_test.cpp)
        target_link_libraries(bedray_history_submission_test PRIVATE bedray_graph_history)
        if(MSVC)
            set_property(TARGET bedray_history_submission_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
    endif()
endif()

option(BEDRAY_GRAPH_VULKAN "Build the experimental Vulkan recording adapter" OFF)
if(BEDRAY_GRAPH_VULKAN)
    set(BEDRAY_VULKAN_HEADERS "" CACHE PATH "Vulkan-Headers include directory")
    if(NOT EXISTS "${BEDRAY_VULKAN_HEADERS}/vulkan/vulkan.h")
        message(FATAL_ERROR "Pass -DBEDRAY_VULKAN_HEADERS=<Vulkan-Headers include directory>")
    endif()
    add_library(bedray_graph_vulkan STATIC src/vulkan_recording.cpp src/core_dispatch.cpp)
    set_property(TARGET bedray_graph_vulkan PROPERTY POSITION_INDEPENDENT_CODE ON)
    target_include_directories(bedray_graph_vulkan PUBLIC include "${BEDRAY_VULKAN_HEADERS}"
        "${CMAKE_CURRENT_LIST_DIR}/../bedray_core/include")
    target_compile_features(bedray_graph_vulkan PUBLIC cxx_std_20)
    target_link_libraries(bedray_graph_vulkan PUBLIC bedray_render_graph)
    if(MSVC)
        set_property(TARGET bedray_graph_vulkan PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
